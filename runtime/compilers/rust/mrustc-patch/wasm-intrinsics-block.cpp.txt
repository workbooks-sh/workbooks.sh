                // ---- Workbooks: wasm32 LLVM intrinsics → clang `__builtin_wasm_*` --------------
                // Lowers the `llvm.wasm.*` intrinsics mrustc otherwise aborts on. The C backend
                // is clang (our clang.wasm targeting wasm32), so the matching clang builtins exist.
                //  * The FUTEX pair (`memory.atomic.wait32/64` + `notify`) is rayon-core's only
                //    blocker — Condvar/LockLatch worker park/wake. Scalar int/ptr args, 1:1 map.
                //  * The SIMD (v128) set: mrustc passes a v128 as a 16-byte struct (`arg0`); clang's
                //    builtins want a clang ext-vector. We bit-cast through a local vector typedef
                //    (`__builtin_memcpy` in/out of `rv`), call the builtin, copy the result back.
                else if( item.m_linkage.name == "llvm.wasm.memory.atomic.wait32" ) {
                    // (ptr: *mut i32, exp: i32, timeout: i64) -> i32
                    m_of << "\treturn __builtin_wasm_memory_atomic_wait32((int*)arg0, arg1, arg2);\n";
                }
                else if( item.m_linkage.name == "llvm.wasm.memory.atomic.wait64" ) {
                    // (ptr: *mut i64, exp: i64, timeout: i64) -> i32
                    m_of << "\treturn __builtin_wasm_memory_atomic_wait64((long long*)arg0, arg1, arg2);\n";
                }
                else if( item.m_linkage.name == "llvm.wasm.memory.atomic.notify" ) {
                    // (ptr: *mut i32, cnt: i32) -> i32
                    m_of << "\treturn __builtin_wasm_memory_atomic_notify((int*)arg0, arg1);\n";
                }
                // ---- wasm SIMD (v128) intrinsics ------------------------------------------------
                else if( item.m_linkage.name.rfind("llvm.wasm.", 0) == 0 )
                {
                    // Local clang ext-vector typedefs for marshalling v128 lanes.
                    m_of
                        << "\ttypedef signed char    v_i8  __attribute__((__vector_size__(16)));\n"
                        << "\ttypedef unsigned char  v_u8  __attribute__((__vector_size__(16)));\n"
                        << "\ttypedef short          v_i16 __attribute__((__vector_size__(16)));\n"
                        << "\ttypedef unsigned short v_u16 __attribute__((__vector_size__(16)));\n"
                        << "\ttypedef int            v_i32 __attribute__((__vector_size__(16)));\n"
                        << "\ttypedef unsigned int   v_u32 __attribute__((__vector_size__(16)));\n"
                        << "\ttypedef long long      v_i64 __attribute__((__vector_size__(16)));\n"
                        << "\ttypedef float          v_f32 __attribute__((__vector_size__(16)));\n"
                        << "\ttypedef double         v_f64 __attribute__((__vector_size__(16)));\n"
                        ;
                    const auto& nm = item.m_linkage.name;
                    // Helper: load arg<N> as vector type V into local <dst>.
                    auto load = [&](const char* dst, const char* vty, int argn) {
                        m_of << "\t" << vty << " " << dst << "; __builtin_memcpy(&" << dst
                             << ", &arg" << argn << ", 16);\n";
                    };
                    auto store_rv = [&](const char* src) {
                        m_of << "\t__builtin_memcpy(&rv, &" << src << ", sizeof(rv));\n\treturn rv;\n";
                    };
                    // bitmask: lane-msb gather → i32. (i8x16 / i16x8 / i32x4 / i64x2)
                    if( nm == "llvm.wasm.bitmask.v16i8" ) {
                        load("a","v_i8",0); m_of << "\treturn __builtin_wasm_bitmask_i8x16(a);\n";
                    }
                    else if( nm == "llvm.wasm.bitmask.v8i16" ) {
                        load("a","v_i16",0); m_of << "\treturn __builtin_wasm_bitmask_i16x8(a);\n";
                    }
                    else if( nm == "llvm.wasm.bitmask.v4i32" ) {
                        load("a","v_i32",0); m_of << "\treturn __builtin_wasm_bitmask_i32x4(a);\n";
                    }
                    else if( nm == "llvm.wasm.bitmask.v2i64" ) {
                        load("a","v_i64",0); m_of << "\treturn __builtin_wasm_bitmask_i64x2(a);\n";
                    }
                    // any_true / all_true: reductions → bool(i32)
                    else if( nm == "llvm.wasm.anytrue.v16i8" || nm == "llvm.wasm.anytrue.v8i16"
                          || nm == "llvm.wasm.anytrue.v4i32" || nm == "llvm.wasm.anytrue.v2i64" ) {
                        load("a","v_i8",0); m_of << "\treturn __builtin_wasm_any_true_v128(a);\n";
                    }
                    else if( nm == "llvm.wasm.alltrue.v16i8" ) {
                        load("a","v_i8",0); m_of << "\treturn __builtin_wasm_all_true_i8x16(a);\n";
                    }
                    else if( nm == "llvm.wasm.alltrue.v8i16" ) {
                        load("a","v_i16",0); m_of << "\treturn __builtin_wasm_all_true_i16x8(a);\n";
                    }
                    else if( nm == "llvm.wasm.alltrue.v4i32" ) {
                        load("a","v_i32",0); m_of << "\treturn __builtin_wasm_all_true_i32x4(a);\n";
                    }
                    else if( nm == "llvm.wasm.alltrue.v2i64" ) {
                        load("a","v_i64",0); m_of << "\treturn __builtin_wasm_all_true_i64x2(a);\n";
                    }
                    // swizzle: i8x16(a, idx) -> i8x16
                    else if( nm == "llvm.wasm.swizzle" ) {
                        load("a","v_i8",0); load("b","v_i8",1);
                        m_of << "\tv_i8 r = __builtin_wasm_swizzle_i8x16(a, b);\n"; store_rv("r");
                    }
                    // narrow (saturating pack): 2× wide → 1× narrow
                    else if( nm == "llvm.wasm.narrow.signed.v16i8.v8i16" ) {
                        load("a","v_i16",0); load("b","v_i16",1);
                        m_of << "\tv_i8 r = __builtin_wasm_narrow_s_i8x16_i16x8(a, b);\n"; store_rv("r");
                    }
                    else if( nm == "llvm.wasm.narrow.unsigned.v16i8.v8i16" ) {
                        load("a","v_i16",0); load("b","v_i16",1);
                        m_of << "\tv_u8 r = __builtin_wasm_narrow_u_i8x16_i16x8(a, b);\n"; store_rv("r");
                    }
                    else if( nm == "llvm.wasm.narrow.signed.v8i16.v4i32" ) {
                        load("a","v_i32",0); load("b","v_i32",1);
                        m_of << "\tv_i16 r = __builtin_wasm_narrow_s_i16x8_i32x4(a, b);\n"; store_rv("r");
                    }
                    else if( nm == "llvm.wasm.narrow.unsigned.v8i16.v4i32" ) {
                        load("a","v_i32",0); load("b","v_i32",1);
                        m_of << "\tv_u16 r = __builtin_wasm_narrow_u_i16x8_i32x4(a, b);\n"; store_rv("r");
                    }
                    // q15mulr saturating: i16x8
                    else if( nm == "llvm.wasm.q15mulr.sat.signed" ) {
                        load("a","v_i16",0); load("b","v_i16",1);
                        m_of << "\tv_i16 r = __builtin_wasm_q15mulr_sat_s_i16x8(a, b);\n"; store_rv("r");
                    }
                    // extadd_pairwise: widen-and-add adjacent lanes
                    else if( nm == "llvm.wasm.extadd.pairwise.signed.v8i16" ) {
                        load("a","v_i8",0);
                        m_of << "\tv_i16 r = __builtin_wasm_extadd_pairwise_i8x16_s_i16x8(a);\n"; store_rv("r");
                    }
                    else if( nm == "llvm.wasm.extadd.pairwise.unsigned.v8i16" ) {
                        load("a","v_u8",0);
                        m_of << "\tv_u16 r = __builtin_wasm_extadd_pairwise_i8x16_u_i16x8(a);\n"; store_rv("r");
                    }
                    else if( nm == "llvm.wasm.extadd.pairwise.signed.v4i32" ) {
                        load("a","v_i16",0);
                        m_of << "\tv_i32 r = __builtin_wasm_extadd_pairwise_i16x8_s_i32x4(a);\n"; store_rv("r");
                    }
                    else if( nm == "llvm.wasm.extadd.pairwise.unsigned.v4i32" ) {
                        load("a","v_u16",0);
                        m_of << "\tv_u32 r = __builtin_wasm_extadd_pairwise_i16x8_u_i32x4(a);\n"; store_rv("r");
                    }
                    // relaxed fma / fnma (f32x4 / f64x2): a*b + c, -(a*b) + c
                    else if( nm == "llvm.wasm.relaxed.madd.v4f32" ) {
                        load("a","v_f32",0); load("b","v_f32",1); load("c","v_f32",2);
                        m_of << "\tv_f32 r = __builtin_wasm_relaxed_madd_f32x4(a, b, c);\n"; store_rv("r");
                    }
                    else if( nm == "llvm.wasm.relaxed.nmadd.v4f32" ) {
                        load("a","v_f32",0); load("b","v_f32",1); load("c","v_f32",2);
                        m_of << "\tv_f32 r = __builtin_wasm_relaxed_nmadd_f32x4(a, b, c);\n"; store_rv("r");
                    }
                    else if( nm == "llvm.wasm.relaxed.madd.v2f64" ) {
                        load("a","v_f64",0); load("b","v_f64",1); load("c","v_f64",2);
                        m_of << "\tv_f64 r = __builtin_wasm_relaxed_madd_f64x2(a, b, c);\n"; store_rv("r");
                    }
                    else if( nm == "llvm.wasm.relaxed.nmadd.v2f64" ) {
                        load("a","v_f64",0); load("b","v_f64",1); load("c","v_f64",2);
                        m_of << "\tv_f64 r = __builtin_wasm_relaxed_nmadd_f64x2(a, b, c);\n"; store_rv("r");
                    }
                    // relaxed laneselect: per-lane bitwise select (mask picks a or b)
                    else if( nm == "llvm.wasm.relaxed.laneselect.v16i8" ) {
                        load("a","v_i8",0); load("b","v_i8",1); load("m","v_i8",2);
                        m_of << "\tv_i8 r = __builtin_wasm_relaxed_laneselect_i8x16(a, b, m);\n"; store_rv("r");
                    }
                    else if( nm == "llvm.wasm.relaxed.laneselect.v8i16" ) {
                        load("a","v_i16",0); load("b","v_i16",1); load("m","v_i16",2);
                        m_of << "\tv_i16 r = __builtin_wasm_relaxed_laneselect_i16x8(a, b, m);\n"; store_rv("r");
                    }
                    else if( nm == "llvm.wasm.relaxed.laneselect.v4i32" ) {
                        load("a","v_i32",0); load("b","v_i32",1); load("m","v_i32",2);
                        m_of << "\tv_i32 r = __builtin_wasm_relaxed_laneselect_i32x4(a, b, m);\n"; store_rv("r");
                    }
                    else if( nm == "llvm.wasm.relaxed.laneselect.v2i64" ) {
                        load("a","v_i64",0); load("b","v_i64",1); load("m","v_i64",2);
                        m_of << "\tv_i64 r = __builtin_wasm_relaxed_laneselect_i64x2(a, b, m);\n"; store_rv("r");
                    }
                    // relaxed swizzle
                    else if( nm == "llvm.wasm.relaxed.swizzle" ) {
                        load("a","v_i8",0); load("b","v_i8",1);
                        m_of << "\tv_i8 r = __builtin_wasm_relaxed_swizzle_i8x16(a, b);\n"; store_rv("r");
                    }
                    else {
                        m_of << "\tassert(!\"Unsupported LLVM wasm intrinsic: " << nm << "\"); abort();\n";
                    }
                }
