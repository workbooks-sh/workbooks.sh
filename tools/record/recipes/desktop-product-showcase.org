* 00:41 eval_js
  Drag a file onto Bookmarks — and it sticks.
  :PROPERTIES:
  :code: (async () => { const file=document.querySelector('.result[draggable="true"], .tile.file[draggable="true"]'); const bm=document.querySelector('[role="group"][aria-label="Bookmarks"]'); if(!file||!bm) return 'missing'; const dt=new DataTransfer(); const fr=file.getBoundingClientRect(); const br=bm.getBoundingClientRect(); const ev=(type,el,x,y)=>el.dispatchEvent(new DragEvent(type,{bubbles:true,cancelable:true,dataTransfer:dt,clientX:x,clientY:y})); ev('dragstart',file,fr.x+fr.width/2,fr.y+fr.height/2); await new Promise(r=>setTimeout(r,60)); ev('dragenter',bm,br.x+br.width/2,br.y+br.height/2); ev('dragover',bm,br.x+br.width/2,br.y+br.height/2); await new Promise(r=>setTimeout(r,60)); ev('drop',bm,br.x+br.width/2,br.y+br.height/2); ev('dragend',file,br.x+br.width/2,br.y+br.height/2); await new Promise(r=>setTimeout(r,700)); return 'dropped'; })()
  :checkpoint: a bookmark tile now sits in the sidebar Bookmarks group (the dashed placeholder is gone)
  :END:
